```dataviewjs
(async () => {
    // SSOT : le registre canonique (Règle 2 du §7) — jamais le frontmatter des specs
    const REG_PATH = "01-Principes_Spécifications/registry_exigences.json";
    const norm = s => String(s || "").trim().toUpperCase();
    const regFile = app.vault.getAbstractFileByPath(REG_PATH);
    if (!regFile) { dv.paragraph("⚠️ Registre canonique introuvable : " + REG_PATH); return; }
    const reg = JSON.parse(await app.vault.read(regFile));
    const exigences = reg.exigences || [];
    const byId = new Map(exigences.map(e => [norm(e.id), e]));
    const totalExgs = exigences.length;

    // Répartition par recette (gouvernance : auto / visuelle / inspection / prevue)
    const parRecette = {};
    for (const e of exigences) {
        const r = (e.recette || "-").split(" ")[0].toLowerCase();
        parRecette[r] = (parRecette[r] || 0) + 1;
    }

    // Scan des tests — matching INSENSIBLE À LA CASSE (leçon reprise : AGE_1 vs age_1)
    let coveredExgs = new Set();
    let orphanFunctionsCount = 0;
    const pyFiles = app.vault.getFiles().filter(f =>
        f.path.includes("03-Application/Test/") && f.path.endsWith(".py"));
    for (const file of pyFiles) {
        const content = await app.vault.read(file);
        (content.match(/EXG-[\w-]+/gi) || []).forEach(id => {
            if (byId.has(norm(id))) coveredExgs.add(norm(id));
        });
        for (const blk of content.split(/(?=def test_)/)) {
            if (!blk.trim().startsWith("def test_")) continue;
            const hits = (blk.match(/EXG-[\w-]+/gi) || []).map(norm).filter(id => byId.has(id));
            if (hits.length === 0) orphanFunctionsCount++;
        }
    }

    const coverageRate = totalExgs > 0 ? ((coveredExgs.size / totalExgs) * 100).toFixed(1) : 0;
    const recetteTxt = Object.entries(parRecette).sort()
        .map(([k, v]) => `${k}: ${v}`).join(" · ");

    dv.paragraph(
`> [!info] **Tableau de Bord Exigences & Traçabilité — source : registre canonique v${(reg.metadata && reg.metadata.spec_driven_version) || "?"}**
> - 📋 **Total Exigences (registre) :** \`${totalExgs}\` — par recette : ${recetteTxt}
> - 🎯 **Couvertes par un test :** \`${coveredExgs.size} (${coverageRate}%)\`
> - ⚠️ **Non couvertes** (visuelle/inspection/prevue incluses) : \`${totalExgs - coveredExgs.size}\`
> - 🧩 **Fonctions de test sans UID registre :** \`${orphanFunctionsCount}\`
> *(Répartition NonReg : 130 passed = plancher courant ; les orphelins vrais restent à relier au registre — cf. section 5.)*`);
})();
```

### 4. Couverture des exigences par des tests
```dataviewjs
(async () => {
    const REG_PATH = "01-Principes_Spécifications/registry_exigences.json";
    const norm = s => String(s || "").trim().toUpperCase();
    const regFile = app.vault.getAbstractFileByPath(REG_PATH);
    if (!regFile) { dv.paragraph("⚠️ Registre canonique introuvable : " + REG_PATH); return; }
    const reg = JSON.parse(await app.vault.read(regFile));

    // Scan tests + NonReg — matching insensible à la casse
    const testMapping = {};   // UID (haut de casse) -> [fichiers]
    const nonRegMapping = {};
    const pyFiles = app.vault.getFiles().filter(f => f.path.endsWith(".py"));
    for (const file of pyFiles) {
        const content = await app.vault.read(file);
        const hasNonRegMark = /non_regression/i.test(content);
        const unique = [...new Set((content.match(/EXG-[\w-]+/gi) || []).map(norm))];
        for (const id of unique) {
            (testMapping[id] = testMapping[id] || []).push(file.name);
            if (hasNonRegMark) nonRegMapping[id] = true;
        }
    }

    const rows = (reg.exigences || []).map(e => {
        const id = norm(e.id), rec = (e.recette || "-").split(" ")[0];
        const testedBy = testMapping[id] ? testMapping[id].join(", ")
            : (rec === "visuelle" || rec === "inspection") ? "— (recette " + rec + ")" : "⚠️ Non couvert";
        return [
            e.spec_source || "-",
            e.id,
            e.domaine || "-",
            e.core ? "Core" : "Non-Core",
            nonRegMapping[id] ? "✅" : "❌",
            (e.titre || "-").replace(/(\r\n|\n|\r)/gm, " "),
            rec,
            testedBy
        ];
    }).sort((a, b) => a[1].localeCompare(b[1]));

    if (rows.length === 0) { dv.paragraph("⚠️ Registre vide."); return; }
    dv.table(["Spec source", "ID Exigence", "Domaine", "Core/Non-Core", "NonReg",
              "Intitulé", "Recette", "Fichier(s) de Test Associé(s)"], rows);
})();
```

### 5. Tests Orphelins

```dataviewjs
(async () => {
    const REG_PATH = "01-Principes_Spécifications/registry_exigences.json";
    const norm = s => String(s || "").trim().toUpperCase();
    const regFile = app.vault.getAbstractFileByPath(REG_PATH);
    if (!regFile) { dv.paragraph("⚠️ Registre canonique introuvable : " + REG_PATH); return; }
    const reg = JSON.parse(await app.vault.read(regFile));
    const validExgs = new Set((reg.exigences || []).map(e => norm(e.id)));

    const pyFiles = app.vault.getFiles().filter(f =>
        f.path.includes("03-Application/Test/") && f.path.endsWith(".py"));
    const rows = [];
    for (const file of pyFiles) {
        const content = await app.vault.read(file);
        const fileHasValidExg = (content.match(/EXG-[\w-]+/gi) || [])
            .some(id => validExgs.has(norm(id)));
        for (const funcBlock of content.split(/(?=def test_)/)) {
            if (!funcBlock.trim().startsWith("def test_")) continue;
            const funcNameMatch = funcBlock.match(/def\s+(test_[a-zA-Z0-9_]+)\s*\(/);
            const funcName = funcNameMatch ? funcNameMatch[1] : "test_inconnu";
            const attached = (funcBlock.match(/EXG-[\w-]+/gi) || [])
                .filter(id => validExgs.has(norm(id)));
            if (attached.length === 0) {
                rows.push([
                    `[[${file.path}|${file.name}]]`,
                    funcName,
                    fileHasValidExg
                        ? "⚠️ UID registre présent au niveau fichier mais pas dans la docstring de la fonction"
                        : "❌ Aucun UID registre dans le fichier (Orphelin Total — à relier au registre, convention 01.1 §4)"
                ]);
            }
        }
    }
    if (rows.length === 0) {
        dv.paragraph("🎉 **Parfait !** Tous les tests sont rattachés à une exigence du registre.");
    } else {
        dv.paragraph(`⚠️ **${rows.length} alerte(s) de traçabilité détectée(s) :**`);
        dv.table(["Fichier de Test", "Fonction de Test", "Statut de Traçabilité"], rows);
    }
})();
```

```dataviewjs
                            domaine: ex.domaine || "—" };
        }

    // ── 2. Scan CODE : chaque annotation EXG-* avec fichier + ligne + contexte

        /\.(py|js|html|css)$/.test(f.path) &&

            const fn = blk.match(/def\s+(test_[\w]+)/);

            for (const m of blk.matchAll(/EXG-[\w]+(?:-[\w]+)*/g))
                (tests[m[0].toUpperCase()] = tests[m[0].toUpperCase()] || []).push(
                    `[[${f.path}|${fn[1]}]]`);
        }
    }
    for (const k in tests) tests[k] = [...new Set(tests[k])];

    // ── 4. Fusion : toutes les exigences P8, ancrage par ancrage
    const allUids = [...new Set([...Object.keys(reg),
        ...Object.keys(impl), ...Object.keys(tests)])]
        .filter(id => /-P8-|GOU_/.test(id)).sort();

    const rows = allUids.map(id => [
        reg[id] ? reg[id].spec : "⚠️ **ABSENT DE LA SPEC**",
        id,
        reg[id] ? reg[id].titre : "—",
        impl[id] ? impl[id].join("<br>") : "— (voir recette visuelle/inspection)",
        tests[id] ? tests[id].join("<br>") : "⚠️ non couvert",

    ]);

    dv.table(["Spécification", "Exigence", "Intitulé",
              "Implémentation (fichier · ligne)", "Test pytest", "Section canvas"],
             rows);

    // ── 5. KPI de dérive
    const noSpec = allUids.filter(id => !reg[id]);
    const noCode = allUids.filter(id => reg[id] && !impl[id] && !tests[id]);
    const noTest = allUids.filter(id => reg[id] && !tests[id]);
    dv.paragraph(
        `> [!info] KPI traçabilité P8 — ${allUids.length} exigences<br>` +
        (noSpec.length  ? `⚠️ Sans spec : ${noSpec.join(", ")}<br>`  : "✅ Toutes déclarées en spec<br>") +
        (noTest.length  ? `⚠️ Sans test : ${noTest.join(", ")}<br>`  : "✅ Toutes testées (ou recette visuelle)<br>") +
        (noCode.length  ? `⚠️ Sans ancrage code : ${noCode.join(", ")}` : "✅ Toutes ancrées dans le code"));
})();
```

**Règle de maintenance (inchangée, pérennisée)** : cette vue est régénérée à l'ouverture — nulle copie manuelle à mettre à jour. Toute nouvelle exigence apparaît dès qu'elle est déclarée en spec **et** annotée `EXG-*` dans le code ; toute évolution de livrable canonique met seulement à jour la table `CANVAS_SECTIONS` en tête de script. Les exigences sans test automatisable portent la mention « recette visuelle » ou « inspection » dans la spec — elles restent traçables.

### 6. Cartographie des Canvas de Conception (lecture humaine)


```dataviewjs
(async () => {
    // ── 0. Configuration : dossiers des canvas de design (à ajuster)
    const CANVAS_FOLDERS = ["00-Projet/phases/", "00-Projet/canvases/"];

    // ── 1. Registre : uid -> {titre, spec, design, domaine}
    const specPages = dv.pages().where(p => p.type === "spec" && p.exigences);
    const reg = {};
    for (const page of specPages)
        for (const ex of page.exigences || []) {
            const id = (ex.uid || ex.id || "").trim().toUpperCase();
            if (id) reg[id] = {
                titre: ex.titre || ex.intitule || "—",
                spec: `[[${page.file.path}|${page.reference || page.file.name}]]`,
                design: (ex.design || "").trim(),
                domaine: ex.domaine || "—"
            };
        }

    // ── 2. Scan CODE et TESTS (pour la colonne couverture) — version corrigée
    const scan = async (pred) => {
        const map = {};  // uid -> Set(files)
        const files = app.vault.getFiles().filter(pred);
        for (const f of files) {
            const content = await app.vault.read(f);
            for (const m of content.matchAll(/EXG-[\w]+(?:-[\w]+)*/g)) {
                (map[m[0].toUpperCase()] = map[m[0].toUpperCase()] || new Set()).add(f.name);
            }
        }
        return map;
    };
    const impl = await scan(f => f.path.startsWith("03-Application/") &&
        /\.(py|js|html|css)$/.test(f.path) &&
        !f.path.includes("/Test/") && !f.path.includes("__pycache__"));
    const tests = await scan(f => f.path.startsWith("03-Application/Test/"));

    // ── 3. Index inverse : section (§xx) -> [uid]
    const sections = {};
    for (const [uid, info] of Object.entries(reg)) {
        if (!info.design) continue;
        for (const sec of info.design.match(/§[\d.]+/g) || []) {
            (sections[sec] = sections[sec] || []).push(uid);
        }
    }

    // ── 4. Lecture des canvas : titres de sections réels
    const canvasFiles = app.vault.getFiles().filter(f =>
        CANVAS_FOLDERS.some(d => f.path.startsWith(d)) && f.extension === "md");
    const sectionTitles = {};  // "§34.2" -> {titre, canvas}
    for (const f of canvasFiles) {
        const lines = (await app.vault.read(f)).split("\n");
        for (let i = 0; i < lines.length; i++) {
            const m = lines[i].match(/^#{1,4}\s*(?:§)?\s*(\d+(?:\.\d+)*)\s*[—–-]?\s*(.*)/);
            if (m) {
                const sec = `§${m[1]}`;
                if (!sectionTitles[sec])
                    sectionTitles[sec] = {
                        titre: m[2].trim() || "(sans titre)",
                        canvas: `[[${f.path}|${f.name}]]`
                    };
            }
        }
    }

    // ── 5. Rendu : un tableau par canvas
    const byCanvas = {};
    const allSecs = new Set([...Object.keys(sections), ...Object.keys(sectionTitles)]);
    for (const sec of allSecs) {
        const info = sectionTitles[sec] ||
            { titre: "⚠️ citée en spec mais introuvable dans les canvas", canvas: "—" };
        (byCanvas[info.canvas] = byCanvas[info.canvas] || []).push({ sec, info });
    }

    for (const [canvas, secs] of Object.entries(byCanvas)) {
        dv.header(3, canvas.replace(/\[\[|\]\]|.*\|/g, ""));
        const rows = secs
            .sort((a, b) => a.sec.localeCompare(b.sec, "fr", { numeric: true }))
            .map(({ sec, info }) => {
                const uids = sections[sec] || [];
                const exgCell = uids.length
                    ? uids.map(u => reg[u]
                        ? `${u} — ${reg[u].titre}`
                        : `${u} ⚠️ (hors registre)`).join("<br>")
                    : "—";
                const specCell = [...new Set(
                    uids.map(u => reg[u] && reg[u].spec).filter(Boolean))].join("<br>") || "—";
                const codeOk = uids.some(u => impl[u]);
                const testOk = uids.length > 0 && uids.every(u => tests[u]);
                return [sec, info.titre, exgCell, specCell,
                        codeOk ? "✅" : "—",
                        testOk ? "✅" : "❌"];
            });
        dv.table(["Section", "Titre", "Exigences ancrées", "Spec", "Code", "Test"], rows);
    }

    // ── 6. KPI d'alerte
    const ghostSecs = Object.keys(sections).filter(s => !sectionTitles[s]);
    if (ghostSecs.length)
        dv.paragraph(`⚠️ **Sections citées en spec mais introuvables :** ${ghostSecs.join(", ")} — à recalage (C6).`);
    const orphanSecs = Object.keys(sectionTitles).filter(s => !sections[s]);
    if (orphanSecs.length)
        dv.paragraph(`ℹ️ **Sections sans exigence ancrée :** ${orphanSecs.length} (design sans contrainte — normal pour les sections méthodo).`);
})();
```

```dataviewjs
    const lines = (await app.vault.read(f)).split("\n");
    const hits = [];
    lines.forEach((line, i) => {
        if (/^#{1,4}\s/.test(line)) {           // tous les headings
            const m = line.match(/^#{1,4}\s*(?:§)?\s*(\d+(?:\.\d+)*)\s*[—–-]?\s*(.*)/);
            hits.push(`${m ? "✅" : "❌"} L${i+1} : ${line.slice(0, 80)}`);
        }
    });
    dv.paragraph(hits.length
        ? `<br>${hits.join("<br>")}`
        : "❌ Aucun heading trouvé dans ce fichier.");
})();
```

### 7. Règles de Sobriété Documentaire (5S, acté 19/11/2026)

Les règles ci-dessous codifient l'aboutissement du 5S sur le corpus documentaire
(Atelier, specs, données). Elles s'imposent à toute nouvelle phase (P9 et
suivantes) ; les nettoyages du corpus ont été exécutés lors de l'étape de
reprise (close le 2026-11-25, CI vert) avec la non-régression portée au
plancher de **130 tests**, jamais à chaud.

**Règle 1 — Une vérité = un fichier.** Chaque donnée (`.ttl`, shapes SHACL,
requêtes SPARQL) vit en un seul exemplaire : c'est la source de vérité,
manuscrite, jamais dupliquée. Les miroirs `.md` et `.json` des graphes
(`DKG_TBox_Master.md`, `.json`, `DKG_ABox_*.md`, etc.) sont **abrogés**
(arbitrage 19/11, exécuté à la reprise ; chemins SSOT supprimés lors de la
régularisation des constantes). Une vue graphique est
**générée à la demande** par `core/mermaid_views.py` (le moteur de vue, marqué
« généré, ne pas éditer ») — jamais manuscrite. Les exportateurs manuels
(`export_phase*_md.py`) sont archivés.

**Règle 2 — Une génération canonique de spécifications.** Les `SPC-*` de
`01-Principes_Spécifications/` sont la génération canonique vivante ;
`registry_exigences.json` est l'index machine unique. Les générations
antérieures (`05-Bin/SPEC-*`, `06-Archives/Specifications_*`) rejoignent les
archives — pas de maintien parallèle. Une exigence formalisée dans une génération
antérieure qui n'existe pas en SPC-* est soit portée, soit explicitement
abrogée ; jamais laissée en double lecture.

**Règle 3 — Rejeu vs moteurs : jamais de copie.** `phases/PhaseN` = SSOT du
rejeu par phase ; `core/` = SSOT des moteurs partagés ; ni l'un ni l'autre ne
copie l'autre — l'`import` fait le pont (aucun module de `core` n'importe
`phases`, les phases et les tests importent `core` et `phases` in situ). Une
« ventilation par duplication » est une dérive : elle crée des exemplaires
divergents (cas `local_host_scanner.py`, dupliqué core/Phase7).

**Règle 4 — Le tri se recette, il ne se improvise pas.** Tout fichier mort,
redondant ou hors-sol identifié par 5S est traité en phase (spec + NonReg),
classé `06-Archives` plutôt que supprimé quand il porte une valeur de rejeu ou
d'histoire. Le grep d'imports réel
(`grep -rn "from core\.\|import core\|from phases\|import phases" --include="*.py"`)
est la preuve canonique de vie d'un module ; le `__pycache__` n'est preuve que
du passé. En fin de phase, aucun module mort ne survit au tree (Sustain).
